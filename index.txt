                d.setDate(now.getDate() + i);

                const dayName = daysMap[d.getDay()];
                const dateNum = d.getDate();
                const monthName = monthsMap[d.getMonth()];

                weekDays.push({
                    index: i,
                    fullLabel: `${dayName}, ${dateNum} ${monthName}`,
                    shortLabel: i === 0 ? "Hari Ini" : dayName,
                    dateFormatted: `${dateNum} ${monthName}`
                });
            }
        }

        // Generate Random Realistic Flight Data with Strict Fleet Logic
        function generateFlightDatabase() {
            flightDatabase = [];
            let counter = 201;

            const airlineList = Object.keys(airlines);
            const airportCodes = Object.keys(airports);

            weekDays.forEach(day => {
                const flightsPerDay = 24;

                for (let i = 0; i < flightsPerDay; i++) {
                    const airlineName = airlineList[Math.floor(Math.random() * airlineList.length)];
                    let origin = airportCodes[Math.floor(Math.random() * airportCodes.length)];
                    let dest = airportCodes[Math.floor(Math.random() * airportCodes.length)];

                    while (dest === origin) {
                        dest = airportCodes[Math.floor(Math.random() * airportCodes.length)];
                    }

                    const depHour = Math.floor(Math.random() * 15) + 6; // 06:00 to 21:00
                    const depMin = Math.floor(Math.random() * 12) * 5;
                    const depTimeStr = `${String(depHour).padStart(2, '0')}:${String(depMin).padStart(2, '0')}`;

                    const durationMinutes = 80 + Math.floor(Math.random() * 70);
                    const arrHour = (depHour + Math.floor(durationMinutes / 60)) % 24;
                    const arrMin = (depMin + (durationMinutes % 60)) % 60;
                    const arrTimeStr = `${String(arrHour).padStart(2, '0')}:${String(arrMin).padStart(2, '0')}`;

                    const status = statusOptions[Math.floor(Math.random() * statusOptions.length)];
                    const flightNum = `${airlines[airlineName].code}-${counter++}`;
                    
                    // Pick aircraft strictly from the airline's assigned fleet
                    const availableFleet = airlineFleetMap[airlineName];
                    const aircraftObj = availableFleet[Math.floor(Math.random() * availableFleet.length)];

                    flightDatabase.push({
                        id: `FL-${counter}`,
                        dayIndex: day.index,
                        airline: airlineName,
                        flightNumber: flightNum,
                        origin: origin,
                        destination: dest,
                        depTime: depTimeStr,
                        arrTime: arrTimeStr,
                        duration: `${Math.floor(durationMinutes / 60)}j ${durationMinutes % 60}m`,
                        terminal: `Terminal ${Math.floor(Math.random() * 3) + 1}`,
                        gate: `G${Math.floor(Math.random() * 18) + 1}`,
                        baggage: `Belt 0${Math.floor(Math.random() * 6) + 1}`,
                        aircraft: aircraftObj.name,
                        aircraftCat: aircraftObj.cat,
                        status: status
                    });
                }
            });
        }

        // Status Badge UI Helper
        function getStatusBadge(status) {
            switch(status) {
                case 'On Time':
                    return `<span class="px-2.5 py-1 rounded-full text-[11px] font-bold bg-emerald-500/10 text-emerald-400 border border-emerald-500/30 inline-flex items-center gap-1.5"><span class="w-1.5 h-1.5 rounded-full bg-emerald-400"></span>On Time</span>`;
                case 'Boarding':
                    return `<span class="px-2.5 py-1 rounded-full text-[11px] font-bold bg-rose-500/15 text-rose-400 border border-rose-500/30 inline-flex items-center gap-1.5 pulse-live"><span class="w-1.5 h-1.5 rounded-full bg-rose-400"></span>Boarding</span>`;
                case 'Delayed':
                    return `<span class="px-2.5 py-1 rounded-full text-[11px] font-bold bg-amber-500/10 text-amber-400 border border-amber-500/30 inline-flex items-center gap-1.5"><span class="w-1.5 h-1.5 rounded-full bg-amber-400"></span>Delayed</span>`;
                case 'Departed':
                    return `<span class="px-2.5 py-1 rounded-full text-[11px] font-bold bg-sky-500/10 text-sky-400 border border-sky-500/30 inline-flex items-center gap-1.5"><span class="w-1.5 h-1.5 rounded-full bg-sky-400"></span>Departed</span>`;
                case 'Cancelled':
                    return `<span class="px-2.5 py-1 rounded-full text-[11px] font-bold bg-slate-800 text-slate-400 border border-slate-700 inline-flex items-center gap-1.5"><span class="w-1.5 h-1.5 rounded-full bg-slate-500"></span>Cancelled</span>`;
                default:
                    return `<span class="px-2.5 py-1 rounded-full text-[11px] font-bold bg-slate-800 text-slate-300">${status}</span>`;
            }
        }

        // Apply Search & Filters
        function applyFilters() {
            const searchVal = document.getElementById('searchInput').value.toLowerCase().trim();
            const airlineVal = document.getElementById('filterAirline').value;
            const originVal = document.getElementById('filterOrigin').value;
            const destVal = document.getElementById('filterDest').value;
            const statusVal = document.getElementById('filterStatus').value;
            const aircraftVal = document.getElementById('filterAircraft').value;

            const filtered = flightDatabase.filter(item => {
                if (item.dayIndex !== selectedDayIndex) return false;
                if (airlineVal !== 'ALL' && item.airline !== airlineVal) return false;
                if (originVal !== 'ALL' && item.origin !== originVal) return false;
                if (destVal !== 'ALL' && item.destination !== destVal) return false;
                if (statusVal !== 'ALL' && item.status !== statusVal) return false;
                if (aircraftVal !== 'ALL' && item.aircraftCat !== aircraftVal) return false;

                if (searchVal) {
                    const origCity = airports[item.origin].city.toLowerCase();
                    const destCity = airports[item.destination].city.toLowerCase();
                    const match = 
                        item.flightNumber.toLowerCase().includes(searchVal) ||
                        item.airline.toLowerCase().includes(searchVal) ||
                        item.aircraft.toLowerCase().includes(searchVal) ||
                        item.origin.toLowerCase().includes(searchVal) ||
                        item.destination.toLowerCase().includes(searchVal) ||
                        origCity.includes(searchVal) ||
                        destCity.includes(searchVal);
                    if (!match) return false;
                }

                return true;
            });

            updateStats();
            renderDesktopTable(filtered);
            renderMobileCards(filtered);

            document.getElementById('flightCountBadge').innerText = `${filtered.length} Flight`;
            
            // Badge text summary
            const activeBadge = document.getElementById('activeFilterBadge');
            let desc = `Penerbangan ${weekDays[selectedDayIndex].shortLabel}`;
            if (originVal !== 'ALL') desc += ` dari ${originVal}`;
            if (destVal !== 'ALL') desc += ` ke ${destVal}`;
            if (aircraftVal !== 'ALL') desc += ` (${aircraftVal})`;
            activeBadge.innerText = desc;

            const emptyEl = document.getElementById('emptyState');
            if (filtered.length === 0) {
                emptyEl.classList.remove('hidden');
            } else {
                emptyEl.classList.add('hidden');
            }
        }

        // Calculate and Update Top Dashboard Stats
        function updateStats() {
            const dayFlights = flightDatabase.filter(f => f.dayIndex === selectedDayIndex);
            const total = dayFlights.length;
            const onTime = dayFlights.filter(f => f.status === 'On Time' || f.status === 'Departed').length;
            const delayed = dayFlights.filter(f => f.status === 'Delayed').length;
            const boarding = dayFlights.filter(f => f.status === 'Boarding').length;

            const onTimeRate = total > 0 ? Math.round((onTime / total) * 100) : 0;

            document.getElementById('statTotal').innerText = total;
            document.getElementById('statOnTime').innerText = onTime;
            document.getElementById('statOnTimeRate').innerText = `${onTimeRate}% Tepat Waktu`;
            document.getElementById('statDelayed').innerText = delayed;
            document.getElementById('statBoarding').innerText = boarding;
        }
